    #include 

int main()
{
    int arr[50];
    int n = 0;
    int choice, pos, value, i;

    while(1)
    {
        printf("\n1. Insert\n2. Delete\n3. Traverse\n4. Search\n5. Update\n6. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
                printf("Enter position and value: ");
                scanf("%d %d", &pos, &value);

                for(i = n; i > pos; i--)
                    arr[i] = arr[i-1];

                arr[pos] = value;
                n++;
                break;

            case 2:
                printf("Enter position to delete: ");
                scanf("%d", &pos);

                for(i = pos; i < n-1; i++)
                    arr[i] = arr[i+1];

                n--;
                break;

            case 3:
                printf("Array elements are:\n");
                for(i = 0; i < n; i++)
                    printf("%d ", arr[i]);
                printf("\n");
                break;

            case 4:
                printf("Enter value to search: ");
                scanf("%d", &value);

                for(i = 0; i < n; i++)
                {
                    if(arr[i] == value)
                    {
                        printf("Element found at index %d\n", i);
                        break;
                    }
                }
                if(i == n)
                    printf("Element not found\n");
                break;

            case 5:
                printf("Enter position and new value: ");
                scanf("%d %d", &pos, &value);

                arr[pos] = value;
                break;

            case 6:
                return 0;

            default:
                printf("Invalid choice\n");
        }
    }
}

      #include 

int main()
{
    int a[10][10], b[10][10];
    int rows, cols;
    int i, j;
    int normal_ops, sparse_ops = 0;

    printf("Enter number of rows and columns: ");
    scanf("%d %d", &rows, &cols);

    printf("Enter elements of first matrix:\n");
    for(i = 0; i < rows; i++)
    {
        for(j = 0; j < cols; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter elements of second matrix:\n");
    for(i = 0; i < rows; i++)
    {
        for(j = 0; j < cols; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    normal_ops = rows * cols;

    for(i = 0; i < rows; i++)
    {
        for(j = 0; j < cols; j++)
        {
            if(a[i][j] != 0)
                sparse_ops++;

            if(b[i][j] != 0)
                sparse_ops++;
        }
    }

    printf("\nOperations in normal matrix addition = %d\n", normal_ops);
    printf("Operations in sparse matrix addition = %d\n", sparse_ops);

    return 0;
}
