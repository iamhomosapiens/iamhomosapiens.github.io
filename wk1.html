int main()
{
    int matrix[10][10];
    int size, row, column;
    int sum = 0;

    printf("Enter the size of the square matrix: ");
    scanf("%d", &size);

    printf("Enter the matrix elements:\n");
    for(row = 0; row < size; row++)
    {
        for(column = 0; column < size; column++)
        {
            scanf("%d", &matrix[row][column]);
        }
    }

    for(row = 0; row < size; row++)
    {
        sum = sum + matrix[row][size - row - 1];
    }

    printf("Sum of right diagonal elements = %d\n", sum);

    return 0;
}

int main()
{
    int matrix[10][10];
    int size, row, column;

    printf("Enter the order of the square matrix: ");
    scanf("%d", &size);

    printf("Enter the matrix elements:\n");
    for(row = 0; row < size; row++)
    {
        for(column = 0; column < size; column++)
        {
            scanf("%d", &matrix[row][column]);
        }
    }

    printf("\nUpper Triangular Matrix:\n");
    for(row = 0; row < size; row++)
    {
        for(column = 0; column < size; column++)
        {
            if(column >= row)
                printf("%d ", matrix[row][column]);
            else
                printf("0 ");
        }
        printf("\n");
    }

    printf("\nLower Triangular Matrix:\n");
    for(row = 0; row < size; row++)
    {
        for(column = 0; column < size; column++)
        {
            if(column <= row)
                printf("%d ", matrix[row][column]);
            else
                printf("0 ");
        }
        printf("\n");
    }

    return 0;
}

    scanf("%d", &n);

    printf("Enter array elements:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < n; i++)
    {
        b[i] = a[n - i - 1];
    }

    printf("\nElements in reversed array:\n");
    for(i = 0; i < n; i++)
    {
        printf("Index = %d  Value = %d  Address = %p\n", i, b[i], &b[i]);
